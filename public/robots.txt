# Do not disallow /_next/static as a whole. Googlebot needs the CSS in
# /_next/static/css/ and the JavaScript in /_next/static/chunks/ to render
# the page. Font files in /_next/static/media/ are woff2. Crawlers are asked
# not to fetch them. Browsers still can. A font response also carries
# X-Robots-Tag: noindex. CSS and JavaScript are not given that header.
Allow: /_next/static/css/
Allow: /_next/static/chunks/
Disallow: /_next/static/media/

# /favicon.ico is the icon beside results. It is allowed, it returns 200 as
# an icon file, it is not in the sitemap, and the response sends
# X-Robots-Tag: noindex so the file URL is not treated as a page.
# /apple-touch-icon.png, /og-image.png, and /mentor.jpg are images, not pages.
# They stay out of the sitemap and are sent with noindex.
Allow: /favicon.ico
